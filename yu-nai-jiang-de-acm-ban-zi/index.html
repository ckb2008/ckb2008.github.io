int qinv(int a, int b = mod - 2) {
    int res = 1;
    for (; b; b >>= 1) {
        if (b & 1) {
            res = res * a % mod;
        }
        a = a * a % mod;
    }
    return res;
}

    read(n);
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= n; ++j) {
            read(a[i][j]);
        }
        a[i][n + i] = 1;
    }

    int fl = 1;
    for (int i = 1; i <= n; ++i) {
        int p = i;
        for (int j = i + 1; j <= n; ++j) {
            if (a[p][i] < a[j][i]) {
                p = j;
            }
        }

        if (p != i) {
            swap(a[i], a[p]);
        }

        if (!a[i][i]) {
            fl = 0;
            break;
        }
        
        int inv = qinv(a[i][i]);
        for (int k = 1; k <= n; ++k) {
            if (i == k) {
                continue;
            }
            int d = a[k][i] * inv % mod;
            for (int j = i; j <= n * 2; ++j) {

            }
        }

            a[i][j] = a[i][j] * inv % mod;

ll inv(ll a, ll p) {
    ll x, y;
    exgcd(a, p, x, y);
    return (x % p + p) % p;
}

map<ll, ll> mp;

ll bsgs(ll a, ll b, ll p) {
    ll A = sqrt(p);
    while (A * A < p) ++A;
    mp.clear();
    for (int i = 0; i <= A; ++i) {
        mp[b] = i;
        b = b * a % p;
    }

    b = 1;
    ll val = qpow(a, A, p);
    for (int i = 0; i <= A; ++i) {
        b = b * val % p;
        if (mp.count(b)) {
            return (i + 1) * A - mp[b];
        }
    }

    return -1;
}

ll exbsgs(ll a, ll b, ll p) {
    a = a % p, b = b % p;
    if (b == 1 || p == 1) {
        return 0;
    }
    if (!a) {
        return b ? -1 : 1;
    }
    if (a == b) {
        return 1;
    }
    ll x, y, g = exgcd(a, p, x, y), k = 0, val = 1;
    for (; g > 1; g = exgcd(a, p, x, y)) {
        if (b % g) {
            return -1;
        }
        ++k;
        b = b / g, p = p / g;
        val = val * (a / g) % p;
        if (val == b) {
            return k;
        }
    }

    ll ans = bsgs(a, b * inv(val, p) % p, p);
    return ans == -1 ? -1 : ans + k;
}

    for (int i = head[x]; i; i = e[i].ne) {

        L[i] = (i - 1) * sqrt(n) + 1, R[i] = i * sqrt(n);

    for (int i = 1; i <= t; ++i)
        for (int j = L[i]; j <= R[i]; ++j)
            pos[j] = i, sum[i] += a[j];
}

void upd(int l, int r, ll x) {
    int p = pos[l], q = pos[r];
    if (p == q) {
        for (int i = l; i <= r; ++i) a[i] += x;
        sum[p] += x * (r - l + 1);
    }
    else {
        for (int i = p + 1; i <= q - 1; ++i) add[i] += x;
        for (int i = l; i <= R[p]; ++i) a[i] += x;
        sum[p] += x * (R[p] - l + 1);
        for (int i = L[q]; i <= r; ++i) a[i] += x;
        sum[q] += x * (r - L[q] + 1);
    }
}

ll ask(int l, int r) {
    int p = pos[l], q = pos[r]; ll ans = 0;
    if (p == q) {
        for (int i = l; i <= r; ++i) ans += a[i];
        ans += add[p] * (r - l + 1);
    } else {
        for (int i = p + 1; i <= q - 1; ++i) 
            ans += sum[i] + add[i] * (R[i] - L[i] + 1);
        for (int i = l; i <= R[p]; ++i) ans += a[i];
        ans += add[p] * (R[p] - l + 1);
        for (int i = L[q]; i <= r; ++i) ans += a[i];
        ans += add[q] * (r - L[q] + 1);
    }
    return ans;
}

void dfs1(int x, int fa) {
    dep[x] = dep[f[x] = fa] + (sz[x] = 1);
    for (int i = head[x]; i; i = e[i].ne) {
        int tn = e[i].to;
        if (tn == fa) continue;

        sz[x] += sz[tn];

    }

int loc[100007], tot, id[100007], top[100007];
void dfs2(int x, int tp) {
    top[x] = tp, id[loc[x] = ++tot] = x;
    if (!son[x]) return;
    dfs2(son[x], tp);
    for (int i = head[x]; i; i = e[i].ne) {
        int tn = e[i].to;
        if (tn == son[x] || tn == f[x]) continue;
        dfs2(tn, tn);
    }

int su[400007], ad[400007];
void pushup(int p) {
    su[p] = (su[p << 1] + su[p << 1 | 1]) % P;
}

void pushdown(int p, int l, int r) {
    if (!ad[p]) return;
    int mid = l + r >> 1;
    ad[p << 1] += ad[p], ad[p << 1 | 1] += ad[p];
    su[p << 1] += ad[p] * (mid - l + 1);
    su[p << 1 | 1] += ad[p] * (r - mid);
    ad[p] = 0;
}

void build(int p, int l, int r) {
    if (l == r) {
        su[p] = a[id[l]] % P;
        return;
    }
    int mid = l + r >> 1;
    build(p << 1, l, mid);
    build(p << 1 | 1, mid + 1, r);
    pushup(p);
}

void modify(int p, int l, int r, int L, int R, int v) {
    if (L <= l && R >= r) {
        (ad[p] += v) %= P;
        (su[p] += v * (r - l + 1)) %= P;
        return;
    }
    pushdown(p, l, r); 
    int mid = l + r >> 1;
    if (L <= mid) modify(p << 1, l, mid, L, R, v);
    if (R > mid) modify(p << 1 | 1, mid + 1, r, L, R, v);
    pushup(p);
}

int query(int p, int l, int r, int L, int R) {
    if (L <= l && R >= r) return su[p] % P;
    pushdown(p, l, r);
    int mid = l + r >> 1, ans = 0;
    if (L <= mid) (ans += query(p << 1, l, mid, L, R)) %= P;
    if (R > mid) (ans += query(p << 1 | 1, mid + 1, r, L, R)) %= P;
    return ans % P;
}

void upd(int x, int y, int v) {
    while (top[x] != top[y]) {
        if (dep[f[top[x]]] < dep[f[top[y]]]) swap(x, y);
        modify(1, 1, tot, loc[top[x]], loc[x], v);
        x = f[top[x]];
    }
    if (dep[x] > dep[y]) swap(x, y);
    modify(1, 1, tot, loc[x], loc[y], v);
}

int sum(int x, int y) {

    while (top[x] != top[y]) {
        if (dep[f[top[x]]] < dep[f[top[y]]]) swap(x, y);
        (res += query(1, 1, tot, loc[top[x]], loc[x])) %= P;
        x = f[top[x]];
    }
    if (dep[x] > dep[y]) swap(x, y);
    (res += query(1, 1, tot, loc[x], loc[y])) %= P;
    return res; 
}

int lca(int x, int y) {
    while (top[x] != top[y]) {
        if (dep[top[x]] < dep[top[y]]) swap(x, y);
        x = f[top[x]];
    }
    return dep[x] < dep[y] ? x : y;
}

    std::vector<int> L(n + 1, 0), R(n + 1, 0), loc(n + 1, 0);
    auto preDfs = [&](auto self, int x, int fa) -> void {
        L[x] = ++dfn;
        loc[dfn] = x;
        sz[x] = 1;
        go(x) {
            if (tn == fa) continue;
            self(self, tn, x);
            sz[x] += sz[tn];
            if (!son[x] || sz[son[x]] < sz[tn]) son[x] = tn;
        }
        R[x] = dfn;
    };

    preDfs(preDfs, 1, -1);

    std::vector<int> cnt(n + 1, 0);
    std::vector<int> res(n + 1, 0);
    int ans = 0, mx = 0;
    auto ad = [&](int u) {
        ++cnt[col[u]];
        if (cnt[col[u]] == mx) ans += col[u];
        else if (cnt[col[u]] > mx)
            mx = cnt[col[u]], ans = col[u];
    };
    auto de = [&](int u) {
        if (cnt[col[u]] == mx) ans -= col[u];
        --cnt[col[u]];
    };
    auto calc = [&]() {
        return ans;
    };

    auto dfs = [&](auto self, int x, int fa, bool ke) -> void {
        go(x) {
            if (tn == fa || tn == son[x]) continue;
            self(self, tn, x, false);
        }

        if (son[x]) self(self, son[x], x, true);
        go(x) {
            if (tn == fa || tn == son[x]) continue;
            rep (i, L[tn], R[tn]) ad(loc[i]);
        }

        ad(x);
        res[x] = calc();

        if (!ke) {
            rep (i, L[x], R[x]) de(loc[i]);
            ans = mx = 0;
        }
    };

    dfs(dfs, 1, -1, false);

    rep (i, 1, n) printf("%lld ", res[i]);
    
    return 0;
}

#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct node {
    int l, r;
    int val, id;

    #define l(p) tr[p].l
    #define r(p) tr[p].r
    #define val(p) tr[p].val
    #define id(p) tr[p].id
} tr[100007 * 300];

int n, m, tot, root[100007];

int insert(int now, int x, int v, int l = 1, int r = 1e5) {
    int p = ++tot;
    tr[p] = tr[now];
    if (l == r) {
        val(p) += v;
        id(p) = l;
        return p;
    }
    int mid = l + r >> 1;
    if (x <= mid) l(p) = insert(l(now), x, v, l, mid);
    else r(p) = insert(r(now), x, v, mid + 1, r);
    if (val(l(p)) >= val(r(p))) val(p) = val(l(p)), id(p) = id(l(p));
    else val(p) = val(r(p)), id(p) = id(r(p));
    return p;
}

int merge(int a, int b, int l = 1, int r = 1e5) {
    if (!a) return b;
    if (!b) return a;
    if (l == r) {
        val(a) += val(b);
        id(a) = l;
        return a;
    }
    int mid = l + r >> 1;
    l(a) = merge(l(a), l(b), l, mid);
    r(a) = merge(r(a), r(b), mid + 1, r);
    if (val(l(a)) >= val(r(a))) val(a) = val(l(a)), id(a) = id(l(a));
    else val(a) = val(r(a)), id(a) = id(r(a));
    return a;
}

struct edge {
    int to, ne;
} e[100007 * 2];

int cnt, dfn, eul[100007 * 2], loc[100007], f[100007], dep[100007], head[100007];

void preDfs(int x, int fa) {
    eul[++dfn] = x, loc[x] = dfn, f[x] = fa, dep[x] = dep[fa] + 1;
    for (int i = head[x]; i; i = e[i].ne) {
        int tn = e[i].to;
        if (tn == fa) continue;
        preDfs(tn, x);
        eul[++dfn] = x;
    }
}

int mn[100007 * 8];

void build(int p, int l, int r) {
    if (l == r) {
        mn[p] = eul[l];
        return;
    }
    int mid = l + r >> 1;
    build(p << 1, l, mid), build(p << 1 | 1, mid + 1, r);
    mn[p] = dep[mn[p << 1]] < dep[mn[p << 1 | 1]] ? mn[p << 1] : mn[p << 1 | 1];
}

int ask(int l, int r, int p = 1, int L = 1, int R = dfn) {
    if (L > r || R < l) return -1;
    if (L >= l && R <= r) return mn[p];
    int mid = L + R >> 1;
    int lans = ask(l, r, p << 1, L, mid);
    int rans = ask(l, r, p << 1 | 1, mid + 1, R);
    if (lans == -1 || rans == -1) return lans + rans + 1;
    return dep[lans] < dep[rans] ? lans : rans;
}

int lca(int x, int y) {
    int l = loc[x], r = loc[y];
    if (l > r) swap(l, r);
    return ask(l, r);
}

int ans[100007];

void dfs(int x, int fa) {
    for (int i = head[x]; i; i = e[i].ne) {
        int tn = e[i].to;
        if (tn == fa || dep[tn] <= dep[x]) continue;
        dfs(tn, x);
        root[x] = merge(root[x], root[tn]);
    }
    if (val(root[x])) ans[x] = id(root[x]);
}

int main() {
    scanf("%d %d", &n, &m);
    for (int i = 1, x, y; i < n; ++i) {
        scanf("%d %d", &x, &y);
        e[++cnt] = {y, head[x]}, head[x] = cnt;
        e[++cnt] = {x, head[y]}, head[y] = cnt;
    }

    preDfs(1, 0);
    build(1, 1, dfn);

    for (int i = 1, x, y, z; i <= m; ++i) {
        scanf("%d %d %d", &x, &y, &z);
        int lc = lca(x, y);
        root[x] = insert(root[x], z, 1), root[y] = insert(root[y], z, 1);
        root[lc] = insert(root[lc], z, -1);
        if (f[lc]) root[f[lc]] = insert(root[f[lc]], z, -1);
    }

    dfs(1, 0);
    for (int i = 1; i <= n; ++i) printf("%d\n", ans[i]);

    return 0;
}

struct randomhash {
     static uint64_t splitmix64(uint64_t x) {
        x += 0x9e3779b97f4a7c15;  
        x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9;
        x = (x ^ (x >> 27)) * 0x94d049bb133111eb;
        return x ^ (x >> 31);
    }
    size_t operator()(uint64_t x) const {
        static const uint64_t FIXED_RANDOM = chrono::steady_clock::now().time_since_epoch().count();
        return splitmix64(x + FIXED_RANDOM);
    }
};

            if (!elemHash.count(elem)) {
                elemHash[elem] = rng();
            }
            return elemHash[elem];
        }

        }
