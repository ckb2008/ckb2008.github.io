B

题意：给定一个长度为 n 序列，每隔十位做一个分组，第 i 组内的数必须完整覆盖 [(i - 1) * 10 + 1, min(i * 10, n)] 内的所有数，如果不满足，输出 No；否则输出 Yes。

直接模拟即可。有个小技巧：把 n 补齐成 10 的倍数，然后扫描第 i 组组内的数的时候直接判断是否在 [(i - 1) * 10 + 1, i * 10] 的范围内即可。

constexpr int N = 107;

int n, a[N];

int main() {
    read(n);
    for (int i = 1; i <= n; ++i) read(a[i]);

    if (n % 10) {
        while (n % 10) a[++n] = n;
    }

    for (int i = 1; i <= n / 10; ++i) {
        int l = (i - 1) * 10, r = i * 10;
        for (int j = (i - 1) * 10 + 1; j <= i * 10; ++j) {
            if (a[j] < l || a[j] > r) {
                puts("No");
                return 0;
            }
        }
    }

    puts("Yes");
    
    return 0;
}

C

题意：给定一个长度为 n 的排列和 q 次询问，每次询问把排列中值为 a[i] 的数移动到末尾。

用一个数组 b 记录值为 val 的数的下标。每次询问的时候，用一个 cnt 变量，cnt 每次增加 1，然后把 b[val] 赋值为 cnt。最后对序列进行排序即可，每次比较权值的大小为 b 中的值。实现上可以传入一个 lambda 函数作为 cmp 的参数。

constexpr int N = 2e5 + 7;

int n, q, a[N], b[N], id[N], idd[N];

int main() {
    read(n, q);
    for (int i = 1; i <= n; ++i) read(a[i]), id[a[i]] = i;
    int pos = n;
    for (int i = 1; i <= q; ++i) {
        read(b[i]);
        id[b[i]] = ++pos;
    }

    for (int i = 1; i <= n; ++i) idd[i] = i;
    sort(idd + 1, idd + n + 1, [&](int x, int y) {
        return id[x] < id[y];
    });

    for (int i = 1; i <= n; ++i) print(idd[i]), putc(' ');
    
    return 0;
}
