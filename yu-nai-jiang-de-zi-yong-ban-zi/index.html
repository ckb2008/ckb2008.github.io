Pollard-Rho 快速分解素因数

namespace prime_fac {

    const int S = 8;

    long long mult_mod(long long a, long long b, long long c) {
        a %= c, b %= c;
        long long ret = 0;
        long long tmp = a;
        while (b) {
            if (b & 1) {
                ret += tmp;
                if (ret > c) ret -= c;
            }
            tmp <<= 1;
            if (tmp > c) tmp -= c;
            b >>= 1;
        }
        return ret;
    }

    long long qow_mod(long long a, long long n, long long _mod) {
        long long ret = 1;
        long long temp = a % _mod;
        while (n) {
            if (n & 1) ret = mult_mod(ret, temp, _mod);
            temp = mult_mod(temp, temp, _mod);
            n >>= 1;
        }
        return ret;
    }

    bool check(long long a, long long n, long long x, long long t) {
        long long ret = qow_mod(a, x, n);
        long long last = ret;
        for (int i = 1; i <= t; i++) {
            ret = mult_mod(ret, ret, n);
            if (ret == 1 && last != 1 && last != n - 1) return true;
            last = ret;
        }
        if (ret != 1) return true;
        return false;
    }

    mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());
    bool Miller_Rabin(long long n) {
        if (n < 2) return false;
        if (n == 2) return true;
        if ((n & 1) == 0) return false;
        long long x = n - 1;
        long long t = 0;
        while ((x & 1) == 0) { x >>= 1; t++; }

        for (int i = 0; i < S; i++) {
            long long a = rng() % (n - 1) + 1;
            if (check(a, n, x, t))
                return false;
        }

        return true;
    }

    long long factor[100];
    int tol;

    long long gcd(long long a, long long b) {
        long long t;
        while (b) {
            t = a;
            a = b;
            b = t % b;
        }
        if (a >= 0) return a;
        return -a;
    }

    long long pollard_rho(long long x, long long c) {
        long long i = 1, k = 2;
        long long x0 = rng() % (x - 1) + 1;
        long long y = x0;
        while (1) {
            i++;
            x0 = (mult_mod(x0, x0, x) + c) % x;
            long long d = gcd(y - x0, x);
            if (d != 1 && d != x) return d;
            if (y == x0) return x;
            if (i == k) { y = x0; k += k; }
        }
    }
    void findfac(long long n, int k) {
        if (n == 1) return;
        if (Miller_Rabin(n)) {
            factor[tol++] = n;
            return;
        }
        long long p = n;
        int c = k;
        while (p >= n) p = pollard_rho(p, c--);
        findfac(p, k);
        findfac(n / p, k);
    }
    vector<int> fac(long long n) {
        tol = 0;
        vector<int>ret;
        findfac(n, 107);
        for (int i = 0; i < tol; i++) ret.push_back(factor[i]);
        return ret;
    }
}

void solve(int l, int r, int L, int R) {
    if (l > r) return;
    if (L == R) {
        for (int i = l; i <= r; ++i) if (q[i].typ) ++ans[L];
        return;
    }
    int mid = L + R >> 1, p1 = 0, p2 = 0;
    for (int i = l; i <= r; ++i) {
        if (q[i].typ) {
            int sum = ask(q[i].r) - ask(q[i].l - 1);

            else q[i].k -= sum, q2[++p2] = q[i];

            if (q[i].id <= mid) add(q[i].l, 1), q1[++p1] = q[i];
            else q2[++p2] = q[i];

    for (int i = 1; i <= p2; ++i) q[l + p1 + i - 1] = q2[i];

    solve(l + p1, l + p1 + p2 - 1, mid + 1, R);
}

    }

        return a == x.a && b == x.b && c == x.c;

void cdq(int l, int r) {
    if (l == r) return;
    int mid = l + r >> 1;
    cdq(l, mid), cdq(mid + 1, r);
    auto func = [&](auto x, auto y) { return x.b != y.b ? x.b < y.b : x.c < y.c; };
    sort(e + l, e + mid + 1, func);
    sort(e + mid + 1, e + r + 1, func);
    int j = l, k = mid + 1;
    for (; k <= r; ++k) {
        for (; j <= mid && e[j].b <= e[k].b; ++j)
            add(e[j].c, e[j].d);
        e[k].res += ask(e[k].c);
    }
    for (int i = l; i < j; ++i) add(e[i].c, -e[i].d);
}

    read(n, k);

线段树合并

给定一棵树，有 m 次询问。每次询问给出一条路径，路径上所有的点上存下一个给定的数字。求每个节点上出现频率最多的数字是什么。

struct node {
    int l, r;
    int val, id;

    #define l(p) tr[p].l
    #define r(p) tr[p].r
    #define val(p) tr[p].val
    #define id(p) tr[p].id
} tr[100007 * 300];

int n, m, tot, root[100007];

int insert(int now, int x, int v, int l = 1, int r = 1e5) {
    int p = ++tot;
    tr[p] = tr[now];
    if (l == r) {
        val(p) += v;
        id(p) = l;
        return p;
    }
    int mid = l + r >> 1;
    if (x <= mid) l(p) = insert(l(now), x, v, l, mid);
    else r(p) = insert(r(now), x, v, mid + 1, r);
    if (val(l(p)) >= val(r(p))) val(p) = val(l(p)), id(p) = id(l(p));
    else val(p) = val(r(p)), id(p) = id(r(p));
    return p;
}

int merge(int a, int b, int l = 1, int r = 1e5) {
    if (!a) return b;
    if (!b) return a;
    if (l == r) {
        val(a) += val(b);
        id(a) = l;
        return a;
    }
    int mid = l + r >> 1;
    l(a) = merge(l(a), l(b), l, mid);
    r(a) = merge(r(a), r(b), mid + 1, r);
    if (val(l(a)) >= val(r(a))) val(a) = val(l(a)), id(a) = id(l(a));
    else val(a) = val(r(a)), id(a) = id(r(a));
    return a;
}

struct edge {
    int to, ne;
} e[100007 * 2];

int cnt, dfn, eul[100007 * 2], loc[100007], f[100007], dep[100007], head[100007];

void preDfs(int x, int fa) {
    eul[++dfn] = x, loc[x] = dfn, f[x] = fa, dep[x] = dep[fa] + 1;
    for (int i = head[x]; i; i = e[i].ne) {
        int tn = e[i].to;
        if (tn == fa) continue;
        preDfs(tn, x);
        eul[++dfn] = x;
    }
}

int mn[100007 * 8];

void build(int p, int l, int r) {
    if (l == r) {
        mn[p] = eul[l];
        return;
    }
    int mid = l + r >> 1;
    build(p << 1, l, mid), build(p << 1 | 1, mid + 1, r);
    mn[p] = dep[mn[p << 1]] < dep[mn[p << 1 | 1]] ? mn[p << 1] : mn[p << 1 | 1];
}

int ask(int l, int r, int p = 1, int L = 1, int R = dfn) {
    if (L > r || R < l) return -1;
    if (L >= l && R <= r) return mn[p];
    int mid = L + R >> 1;
    int lans = ask(l, r, p << 1, L, mid);
    int rans = ask(l, r, p << 1 | 1, mid + 1, R);
    if (lans == -1 || rans == -1) return lans + rans + 1;
    return dep[lans] < dep[rans] ? lans : rans;
}

int lca(int x, int y) {
    int l = loc[x], r = loc[y];
    if (l > r) swap(l, r);
    return ask(l, r);
}

int ans[100007];

void dfs(int x, int fa) {
    for (int i = head[x]; i; i = e[i].ne) {
        int tn = e[i].to;
        if (tn == fa || dep[tn] <= dep[x]) continue;
        dfs(tn, x);
        root[x] = merge(root[x], root[tn]);
    }
    if (val(root[x])) ans[x] = id(root[x]);
}

int main() {
    scanf("%d %d", &n, &m);
    for (int i = 1, x, y; i < n; ++i) {
        scanf("%d %d", &x, &y);
        e[++cnt] = {y, head[x]}, head[x] = cnt;
        e[++cnt] = {x, head[y]}, head[y] = cnt;
    }

    preDfs(1, 0);
    build(1, 1, dfn);

    for (int i = 1, x, y, z; i <= m; ++i) {
        scanf("%d %d %d", &x, &y, &z);
        int lc = lca(x, y);
        root[x] = insert(root[x], z, 1), root[y] = insert(root[y], z, 1);
        root[lc] = insert(root[lc], z, -1);
        if (f[lc]) root[f[lc]] = insert(root[f[lc]], z, -1);
    }

    dfs(1, 0);
    for (int i = 1; i <= n; ++i) printf("%d\n", ans[i]);

    return 0;
}

#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int n, m, B, res, col[50007], cnt[50007];

struct Q {
    int id, l, r;
    bool operator<(Q x) const {
        return l / B == x.l / B ? (r == x.r ? 0 : ((l / B) & 1) ^ (r < x.r)) : l < x.l;
    }
} que[50007];

struct A {
    ll p, q;
    string k;
} ans[50007];

void move(int val, int sign) {
    sign ? res += cnt[val]++ : res -= --cnt[val];
}

int main() {
    scanf("%d %d", &n, &m);
    B = sqrt(m);
    for (int i = 1; i <= n; ++i) scanf("%d", &col[i]);
    for (int i = 1; i <= m; ++i) {
        int& l = que[i].l;
        int& r = que[i].r;
        int& id = que[i].id;
        scanf("%d %d", &l, &r);
        id = i;
    }

    sort(que + 1, que + m + 1);
    for (int i = 1, l = 1, r = 0; i <= m; ++i) {
        int L = que[i].l, R = que[i].r;
        if (L == R) {
            ans[que[i].id] = {0, 1};
            continue;
        }
        while (l > L) move(col[--l], 1);
        while (r < R) move(col[++r], 1);
        while (l < L) move(col[l++], 0);
        while (r > R) move(col[r--], 0);
        ans[que[i].id] = {res, 1ll * (r - l + 1) * (r - l) / 2};
    }

    for (int i = 1; i <= m; ++i) {
        ll tmpP = ans[que[i].id].p, tmpQ = ans[que[i].id].q;

        if (!tmpP) {
            ans[que[i].id].k = "0/1";
            continue;
        }

        ll g = __gcd(tmpP, tmpQ);
        tmpP /= g, tmpQ /= g;
        ans[que[i].id].k = to_string(tmpP) + "/" + to_string(tmpQ);
    }

    for (int i = 1; i <= m; ++i) printf("%s\n", &ans[i].k[0]);

    return 0;
}

            return curhash ^ getelemhash(elem);
        }
};

void solve(){
    SetHasher sh;
    int n; cin>>n;
    vector<int> a(n + 1, 0), b(n + 1, 0),c(n + 1,0);
    vector<int> ma, mb, mc;
    ma.reserve(n);
    mb.reserve(n);
    mc.reserve(n);
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) cin >> b[i];
    for (int i = 1; i <= n; i++) cin >> c[i];
    int cnt=0;
    ull ha = 0, hb = 0, hc = 0;
    for (int i = 1; i <= n; i++){
        ha = sh.add(ha, a[i]);
        hb = sh.add(hb, b[i]);
        hc = sh.add(hc, c[i]);
        if(ha == hb || ha == hc || hb == hc){
            cnt++;
        }
    }
    cout << cnt << '\n';
}
